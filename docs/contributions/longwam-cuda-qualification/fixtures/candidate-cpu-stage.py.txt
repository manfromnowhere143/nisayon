"""Opt-in full-checkpoint CUDA lifecycle and streaming VAE checks, without downloads.

Set LONGWAM_NATIVE_CHECKPOINT to a converted IDM or COD checkpoint to run.
"""

import os
import threading
from concurrent.futures import ThreadPoolExecutor
from typing import Any

import pytest
import torch
from lerobot.configs import PreTrainedConfig
from lerobot.policies.longwam.modeling_longwam import LongWAMPolicy
from lerobot.policies.longwam.streaming_longwam import StreamingVAE

CHECKPOINT = os.environ.get("LONGWAM_NATIVE_CHECKPOINT")
pytestmark = pytest.mark.skipif(
    not CHECKPOINT or not torch.cuda.is_available(), reason="local CUDA checkpoint required"
)


@pytest.fixture(scope="module")
def policy():
    cfg = PreTrainedConfig.from_pretrained(CHECKPOINT)
    cfg.device, cfg.execution, cfg.trigger_stride, cfg.streaming_vae = "cpu", "async", 5, True
    with LongWAMPolicy.from_pretrained(CHECKPOINT, config=cfg).to("cuda").eval() as model:
        cfg.device = "cuda"
        verify_staged_weights(model)
        yield model


@pytest.fixture
def observation(policy):
    cfg = policy.config
    generator = torch.Generator().manual_seed(7)
    batch = {
        key: torch.rand(1, 3, *cfg.camera_size, generator=generator).to("cuda")
        for key in cfg.camera_keys
    }
    batch.update(
        {
            "observation.state": torch.zeros(1, cfg.proprio_dim, device="cuda"),
            "task": ["pick up the bowl"],
        }
    )
    return batch


def test_full_weight_streaming_vae_matches_window(policy, observation):
    policy.close()
    core, cfg = policy.model, policy.config
    frame = policy._images(observation, temporal=False)[0]
    frames = [
        (frame + i * 0.02).clamp(-1, 1) for i in range(cfg.past_obs_steps // cfg.video_stride + 1)
    ]
    window = torch.stack(frames, dim=1).unsqueeze(0)
    with torch.no_grad():
        expected = core._encode_video_latents(window, tiled=False)
        session = StreamingVAE(core, past_steps=cfg.past_obs_steps, sample_stride=cfg.video_stride)
        session.begin(cfg.past_obs_steps, [(0, frames[0])])
        for step in range(cfg.video_stride, cfg.past_obs_steps + 1, cfg.video_stride):
            session.feed(step, frames[step // cfg.video_stride])
        actual = session.finish(cfg.past_obs_steps)
        error = (expected.float() - actual.float()).abs().max().item()
        print(f"FULL_VAE_MAX_ABS_ERROR={error}", flush=True)
        torch.testing.assert_close(actual, expected, atol=0, rtol=0)
        session.reset()


@pytest.mark.parametrize(
    "transition", ["late_handoff", "reset_pending", "reset_pending_changed_observation"]
)
def test_full_weight_async_pending_transition(
    policy: LongWAMPolicy,
    observation: dict[str, Any],
    monkeypatch: pytest.MonkeyPatch,
    transition: str,
) -> None:
    check_async_pending_transition(policy, observation, monkeypatch, transition)


def check_async_pending_transition(
    policy: LongWAMPolicy,
    observation: dict[str, Any],
    monkeypatch: pytest.MonkeyPatch,
    transition: str,
) -> None:
    """Check native lifecycle assertions; CPU controls separately exercise their sensitivity."""
    policy.close()
    entered, release = threading.Event(), threading.Event()
    chunks, inputs = [], []
    original = policy.model.infer_joint_ar

    def delayed(**kwargs: Any) -> dict[str, Any]:
        worker = policy._runtime.worker
        inputs.append(
            {
                "episode": worker.episode,
                "anchor": worker.last_step,
                "state": kwargs["proprio"].detach().cpu().clone(),
                "image": kwargs["input_image"].detach().cpu().clone(),
            }
        )
        if len(chunks) == 1:
            entered.set()
            if not release.wait(60):
                raise TimeoutError("Lifecycle test did not release the full-model inference")
        result = original(**kwargs)
        chunks.append(result["action"].detach().float().cpu().clone())
        return result

    monkeypatch.setattr(policy.model, "infer_joint_ar", delayed)

    def at_step(step: int) -> dict[str, Any]:
        batch = dict(observation)
        for key in policy.config.camera_keys:
            batch[key] = (observation[key] + step * 0.01).clamp(0, 1)
        batch["observation.state"] = observation["observation.state"] + step * 0.03
        return batch

    next_batch = at_step(10 if transition != "reset_pending" else 0)
    expected_state = next_batch["observation.state"].detach().cpu().clone()
    expected_image = policy._images(next_batch, temporal=False).detach().cpu().clone()
    try:
        first = policy.select_action(at_step(0)).cpu()
        last = 10 if transition == "late_handoff" else 6
        for step in range(1, last):
            policy.select_action(at_step(step))
        assert entered.wait(30)
        if transition != "late_handoff":
            policy.reset()
        with ThreadPoolExecutor(max_workers=1) as caller:
            future = caller.submit(policy.select_action, next_batch)
            try:
                with pytest.raises(TimeoutError):
                    future.result(timeout=0.1)
            finally:
                release.set()
            actual = future.result(120).cpu()
        if transition == "late_handoff":
            assert not torch.equal(chunks[0][5], chunks[1][5]), (
                "Fixture must distinguish old and new chunks"
            )
            torch.testing.assert_close(actual[0], chunks[1][5], atol=0, rtol=0)
            assert policy._runtime.metrics["late_handoffs"] >= 1
        else:
            assert len(inputs) == len(chunks) == 3, "Reset must execute one fresh inference"
            assert [(value["episode"], value["anchor"]) for value in inputs] == [
                (1, 0),
                (1, 5),
                (2, 0),
            ]
            torch.testing.assert_close(
                inputs[2]["state"],
                expected_state,
                atol=0,
                rtol=0,
                msg="Reset inference used stale state",
            )
            torch.testing.assert_close(
                inputs[2]["image"],
                expected_image,
                atol=0,
                rtol=0,
                msg="Reset inference used stale image",
            )
            torch.testing.assert_close(
                actual[0],
                chunks[2][0].to(actual.dtype),
                atol=0,
                rtol=0,
                msg="Reset action must match fresh inference",
            )
            if transition == "reset_pending":
                torch.testing.assert_close(actual, first, atol=0, rtol=0)
            assert policy._runtime.episode == 2
        assert torch.isfinite(actual).all()
    finally:
        release.set()
        policy.close()
    assert policy._runtime is None
    assert not any(t.name.startswith("longwam") for t in threading.enumerate())


def verify_staged_weights(model: LongWAMPolicy) -> None:
    import json
    from pathlib import Path

    from safetensors import safe_open

    state = model.state_dict()
    checked = 0
    total_bytes = 0
    with safe_open(Path(CHECKPOINT) / "model.safetensors", framework="pt", device="cpu") as saved:
        for name in saved.keys():
            actual = state[name]
            expected = saved.get_tensor(name)
            assert actual.is_cuda, name
            assert actual.dtype == expected.dtype, name
            torch.testing.assert_close(
                actual.detach().cpu(), expected, atol=0, rtol=0, equal_nan=False
            )
            checked += 1
            total_bytes += expected.numel() * expected.element_size()
        aliases = {
            name: target
            for name, target in (saved.metadata() or {}).items()
            if name in state and target in state
        }
        for name, target in aliases.items():
            left, right = state[name], state[target]
            assert left.data_ptr() == right.data_ptr(), (name, target)
            assert left.shape == right.shape and left.stride() == right.stride(), (name, target)
    assert model.model.device.type == "cuda"
    assert all(value.is_cuda for value in model.parameters())
    Path(os.environ["N063_MODEL_LOAD_RECORD"]).write_text(
        json.dumps(
            {
                "loading": "native CPU from_pretrained followed by native Module.to(cuda)",
                "strict": True,
                "serialized_tensors_compared": checked,
                "serialized_tensor_bytes": total_bytes,
                "shared_aliases_checked": len(aliases),
                "all_serialized_values_exact": True,
                "all_parameters_cuda": True,
                "config_device": model.config.device,
                "core_device": str(model.model.device),
                "core_dtype": str(model.model.torch_dtype),
                "cuda_allocated_bytes": torch.cuda.memory_allocated(),
                "cuda_max_allocated_bytes": torch.cuda.max_memory_allocated(),
            },
            indent=2,
        )
        + "\n"
    )
